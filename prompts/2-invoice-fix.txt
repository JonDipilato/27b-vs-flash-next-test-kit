# Fix two bugs in invoice.html

This is a contractor invoice calculator. Two things are wrong with it. Find them in the code and fix them.

**Bug 1 — the invoice can go negative.**
Put a discount that's bigger than the line amount (for example, a line of 1 x 3200 with a discount of 99999) and the line total, the subtotal, and the grand total all go negative. An invoice should never total less than zero. A discount must never take a line below zero.

**Bug 2 — the numbers don't add up.**
Add one line item with quantity 3 and unit price 33.335, and set the tax rate to 8.25%. It displays subtotal $100.00, tax $8.25, and total $108.26. But 100.00 + 8.25 is 108.25, not 108.26. The money shown must always add up exactly.

Also: format money with thousands separators, so $24500.00 shows as $24,500.00.

Do not change the visual design, the colors, or the layout. Only fix the logic and the number formatting.


Here is the complete invoice.html. Output ONLY the complete fixed invoice.html file, nothing else.

<script>
  const grid = document.getElementById('grid');
  const empty = document.getElementById('empty');
  const addBtn = document.getElementById('addBtn');
  const clearAll = document.getElementById('clearAll');
  const printBtn = document.getElementById('printBtn');
  const taxInput = document.getElementById('taxRate');
  const taxPct = document.getElementById('taxPct');
  const subtotalEl = document.getElementById('subtotal');
  const taxAmountEl = document.getElementById('taxAmount');
  const totalEl = document.getElementById('total');

  const rows = [];
  let idCounter = 0;

  function makeRow(data = {}) {
    const id = ++idCounter;
    const r = {
      id,
      desc: data.desc || '',
      qty: data.qty != null ? data.qty : 1,
      price: data.price || 0,
      discount: data.discount || 0,
    };
    rows.push(r);

    const desc = document.createElement('input');
    desc.type = 'text';
    desc.className = 'field';
    desc.value = r.desc;
    desc.placeholder = 'e.g. Metal roof panels';
    desc.addEventListener('input', () => { r.desc = desc.value; render(); });

    const qty = document.createElement('input');
    qty.type = 'number';
    qty.className = 'field field-num';
    qty.min = '0';
    qty.step = '1';
    qty.value = r.qty;
    qty.addEventListener('input', () => { r.qty = parseFloat(qty.value) || 0; render(); });

    const price = document.createElement('input');
    price.type = 'number';
    price.className = 'field field-num';
    price.min = '0';
    price.step = '0.01';
    price.value = r.price;
    price.addEventListener('input', () => { r.price = parseFloat(price.value) || 0; render(); });

    const discount = document.createElement('input');
    discount.type = 'number';
    discount.className = 'field field-num';
    discount.min = '0';
    discount.step = '0.01';
    discount.value = r.discount;
    discount.placeholder = '0.00';
    discount.addEventListener('input', () => { r.discount = parseFloat(discount.value) || 0; render(); });

    const lineTotal = document.createElement('div');
    lineTotal.className = 'line-total';
    lineTotal.textContent = '$0.00';

    const remove = document.createElement('button');
    remove.className = 'btn btn-ghost';
    remove.textContent = '×';
    remove.title = 'Remove line item';
    remove.addEventListener('click', () => {
      const idx = rows.findIndex(x => x.id === id);
      if (idx > -1) rows.splice(idx, 1);
      removeRowDom(r.id, [desc, qty, price, discount, lineTotal, remove]);
      render();
    });

    grid.appendChild(desc);
    grid.appendChild(qty);
    grid.appendChild(price);
    grid.appendChild(discount);
    grid.appendChild(lineTotal);
    grid.appendChild(remove);

    r.dom = [desc, qty, price, discount, lineTotal, remove];
    r.lineTotalEl = lineTotal;

    // Focus the new row's description
    desc.focus();
    return r;
  }

  function removeRowDom(id, domEls) {
    // Remove DOM nodes belonging to this row
    domEls.forEach(el => el.parentNode && el.parentNode.removeChild(el));
  }

  function render() {
    // Empty state
    if (rows.length === 0) {
      empty.style.display = '';
    } else {
      empty.style.display = 'none';
    }

    let subtotal = 0;
    rows.forEach(r => {
      const line = (r.qty * r.price) - r.discount;
      r.lineTotalEl.textContent = fmt(line);
      subtotal += line;
    });

    const taxRate = parseFloat(taxInput.value) || 0;
    const taxAmount = subtotal * (taxRate / 100);
    const total = subtotal + taxAmount;

    subtotalEl.textContent = fmt(subtotal);
    taxAmountEl.textContent = fmt(taxAmount);
    totalEl.textContent = fmt(total);
    taxPct.textContent = (taxRate).toFixed(2).replace(/\.00$/, '');
  }

  function fmt(n) {
    return '$' + n.toFixed(2);
  }

  addBtn.addEventListener('click', () => makeRow());
  clearAll.addEventListener('click', () => {
    rows.forEach(r => removeRowDom(r.id, r.dom));
    rows.length = 0;
    render();
  });
  taxInput.addEventListener('input', render);
  printBtn.addEventListener('click', () => window.print());

  // Seed a couple of sample rows so the page looks alive
  makeRow({ desc: 'Metal roof panels', qty: 1, price: 24500, discount: 0 });
  makeRow({ desc: 'Gutter replacement', qty: 1, price: 3200, discount: 200 });
  makeRow({ desc: 'Climb & rigging (labor)', qty: 6, price: 180, discount: 0 });
  render();
</script>
